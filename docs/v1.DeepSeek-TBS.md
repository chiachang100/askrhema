# engine/indexer.py
"""
Bible indexing engine for TBS.
Handles loading Bible data, generating embeddings, and building BM25 indices.
"""

import json
import logging
from pathlib import Path
from typing import List, Dict, Any, Optional, Tuple
import hashlib

import numpy as np
from sentence_transformers import SentenceTransformer
from qdrant_client import QdrantClient
from qdrant_client.models import (
    VectorParams, 
    Distance, 
    PointStruct,
    PayloadSchemaType,
    CreateCollection
)
from rank_bm25 import BM25Okapi
import nltk
from nltk.tokenize import word_tokenize

from config import config

# Setup logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class BibleIndexer:
    """
    Handles indexing of Bible verses for both dense (vector) and sparse (BM25) search.
    """
    
    def __init__(
        self, 
        data_path: Optional[Path] = None,
        embedding_model: str = config.EMBEDDING_MODEL,
        collection_name: str = config.QDRANT_COLLECTION
    ):
        """
        Initialize the Bible indexer.
        
        Args:
            data_path: Path to the Bible JSON data file
            embedding_model: Name of the sentence-transformers model to use
            collection_name: Name of the Qdrant collection
        """
        self.data_path = data_path or config.BIBLE_DATA_PATH
        self.collection_name = collection_name
        self.embedding_model_name = embedding_model
        
        # Initialize components
        self._initialize_models()
        self._initialize_qdrant()
        
        # Cache for BM25 index
        self._bm25_index = None
        self._bm25_corpus = None
        self._verse_ids = None
        
    def _initialize_models(self):
        """Initialize the embedding model."""
        logger.info(f"Loading embedding model: {self.embedding_model_name}")
        self.embedding_model = SentenceTransformer(self.embedding_model_name)
        logger.info("Embedding model loaded successfully")
        
    def _initialize_qdrant(self):
        """Initialize the Qdrant client."""
        try:
            # Using in-memory mode for simplicity
            self.qdrant_client = QdrantClient(":memory:")
            logger.info("Qdrant initialized in-memory mode")
        except Exception as e:
            logger.error(f"Failed to initialize Qdrant: {e}")
            raise
            
    def load_bible_data(self) -> List[Dict[str, Any]]:
        """
        Load Bible data from JSON file.
        
        Returns:
            List of verse dictionaries
        """
        try:
            with open(self.data_path, 'r', encoding='utf-8') as f:
                verses = json.load(f)
            logger.info(f"Loaded {len(verses)} verses from {self.data_path}")
            return verses
        except FileNotFoundError:
            logger.error(f"Data file not found: {self.data_path}")
            raise
        except json.JSONDecodeError as e:
            logger.error(f"Invalid JSON in data file: {e}")
            raise
            
    def prepare_corpus(self, verses: List[Dict[str, Any]]) -> Tuple[List[str], List[str]]:
        """
        Prepare the corpus for BM25 indexing.
        
        Args:
            verses: List of verse dictionaries
            
        Returns:
            Tuple of (tokenized_corpus, verse_ids)
        """
        # Ensure nltk data is downloaded
        try:
            nltk.data.find('tokenizers/punkt')
        except LookupError:
            nltk.download('punkt')
            
        corpus = []
        verse_ids = []
        
        for verse in verses:
            # Combine book, chapter, verse, and text for better context
            text = f"{verse['book']} {verse['chapter']}:{verse['verse']} {verse['text']}"
            corpus.append(text)
            verse_ids.append(str(verse['id']))
            
        # Tokenize for BM25
        tokenized_corpus = [word_tokenize(text.lower()) for text in corpus]
        
        return tokenized_corpus, verse_ids
        
    def build_bm25_index(self, verses: List[Dict[str, Any]]):
        """
        Build the BM25 index from verses.
        
        Args:
            verses: List of verse dictionaries
        """
        tokenized_corpus, verse_ids = self.prepare_corpus(verses)
        
        logger.info(f"Building BM25 index for {len(tokenized_corpus)} verses")
        self._bm25_index = BM25Okapi(tokenized_corpus)
        self._bm25_corpus = tokenized_corpus
        self._verse_ids = verse_ids
        
        logger.info("BM25 index built successfully")
        
    def get_embedding(self, text: str) -> np.ndarray:
        """
        Get embedding vector for text.
        
        Args:
            text: Text to embed
            
        Returns:
            Embedding vector as numpy array
        """
        embedding = self.embedding_model.encode(text)
        return embedding
        
    def create_qdrant_collection(self, vector_size: int = config.VECTOR_SIZE):
        """
        Create Qdrant collection for vector storage.
        
        Args:
            vector_size: Size of embedding vectors
        """
        try:
            # Check if collection exists
            collections = self.qdrant_client.get_collections()
            collection_names = [c.name for c in collections.collections]
            
            if self.collection_name not in collection_names:
                logger.info(f"Creating Qdrant collection: {self.collection_name}")
                
                # Create collection with payload schema
                self.qdrant_client.create_collection(
                    collection_name=self.collection_name,
                    vectors_config=VectorParams(
                        size=vector_size,
                        distance=Distance.COSINE
                    )
                )
                
                # Set up payload schema for filtering
                self.qdrant_client.create_payload_index(
                    collection_name=self.collection_name,
                    field_name="book",
                    field_schema=PayloadSchemaType.KEYWORD
                )
                
                self.qdrant_client.create_payload_index(
                    collection_name=self.collection_name,
                    field_name="testament",
                    field_schema=PayloadSchemaType.KEYWORD
                )
                
                logger.info("Qdrant collection created successfully")
            else:
                logger.info(f"Collection {self.collection_name} already exists")
                
        except Exception as e:
            logger.error(f"Failed to create Qdrant collection: {e}")
            raise
            
    def index_verses(self, verses: List[Dict[str, Any]]):
        """
        Index verses in Qdrant with embeddings.
        
        Args:
            verses: List of verse dictionaries
        """
        # Build BM25 index first
        self.build_bm25_index(verses)
        
        # Create Qdrant collection
        self.create_qdrant_collection()
        
        # Prepare points for Qdrant
        points = []
        for verse in verses:
            text = f"{verse['book']} {verse['chapter']}:{verse['verse']} {verse['text']}"
            embedding = self.get_embedding(text)
            
            point = PointStruct(
                id=verse['id'],
                vector=embedding.tolist(),
                payload={
                    'id': verse['id'],
                    'book': verse['book'],
                    'chapter': verse['chapter'],
                    'verse': verse['verse'],
                    'text': verse['text'],
                    'testament': verse['testament'],
                    'category': verse.get('category', 'Unknown')
                }
            )
            points.append(point)
            
        # Upsert points to Qdrant
        if points:
            logger.info(f"Indexing {len(points)} verses in Qdrant")
            self.qdrant_client.upsert(
                collection_name=self.collection_name,
                points=points
            )
            logger.info("Vector indexing completed")
            
    def index_from_json(self) -> int:
        """
        Main indexing function: loads data and indexes it.
        
        Returns:
            Number of verses indexed
        """
        logger.info("Starting Bible indexing process")
        
        # Load data
        verses = self.load_bible_data()
        if not verses:
            logger.warning("No verses found to index")
            return 0
            
        # Index verses
        self.index_verses(verses)
        
        logger.info(f"Successfully indexed {len(verses)} verses")
        return len(verses)
        
    def get_bm25_index(self):
        """Get the BM25 index."""
        if self._bm25_index is None:
            # Try to build from data
            verses = self.load_bible_data()
            self.build_bm25_index(verses)
        return self._bm25_index
        
    def get_bm25_corpus(self):
        """Get the BM25 corpus."""
        if self._bm25_corpus is None:
            verses = self.load_bible_data()
            self.build_bm25_index(verses)
        return self._bm25_corpus
        
    def get_verse_ids(self):
        """Get the verse IDs for BM25."""
        if self._verse_ids is None:
            verses = self.load_bible_data()
            self.build_bm25_index(verses)
        return self._verse_ids

# engine/hybrid_search.py
"""
Hybrid search engine combining BM25 and vector search with RRF fusion.
"""

import logging
from typing import List, Dict, Any, Optional, Tuple
from collections import defaultdict

import numpy as np
from sentence_transformers import SentenceTransformer
from qdrant_client import QdrantClient
from qdrant_client.models import Filter, FieldCondition, MatchValue
import streamlit as st
import nltk
from nltk.tokenize import word_tokenize

from config import config
from .indexer import BibleIndexer

# Setup logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class HybridSearchEngine:
    """
    Hybrid search engine combining BM25 keyword search and dense vector search
    with Reciprocal Rank Fusion (RRF).
    """
    
    def __init__(
        self,
        embedding_model: str = config.EMBEDDING_MODEL,
        collection_name: str = config.QDRANT_COLLECTION,
        rrf_k: int = config.RRF_K
    ):
        """
        Initialize the hybrid search engine.
        
        Args:
            embedding_model: Name of the sentence-transformers model
            collection_name: Name of the Qdrant collection
            rrf_k: RRF ranking constant
        """
        self.embedding_model_name = embedding_model
        self.collection_name = collection_name
        self.rrf_k = rrf_k
        
        # Initialize components
        self._initialize_models()
        self._initialize_qdrant()
        self._initialize_indexer()
        
    @st.cache_resource
    def _initialize_models(self):
        """Initialize the embedding model with caching."""
        logger.info(f"Loading embedding model: {self.embedding_model_name}")
        model = SentenceTransformer(self.embedding_model_name)
        logger.info("Embedding model loaded successfully")
        return model
        
    @st.cache_resource
    def _initialize_qdrant(self):
        """Initialize the Qdrant client with caching."""
        try:
            client = QdrantClient(":memory:")
            logger.info("Qdrant initialized in-memory mode")
            return client
        except Exception as e:
            logger.error(f"Failed to initialize Qdrant: {e}")
            raise
            
    @st.cache_resource
    def _initialize_indexer(self):
        """Initialize the Bible indexer with caching."""
        return BibleIndexer()
        
    def __init_subclass__(cls):
        """Handle initialization after class creation."""
        # We need to handle this differently since we're using cached resources
        pass
        
    def __init__(self, *args, **kwargs):
        """Initialize the search engine with caching."""
        self.embedding_model = self._initialize_models()
        self.qdrant_client = self._initialize_qdrant()
        self.indexer = self._initialize_indexer()
        
        # Ensure BM25 index is built
        if self.indexer._bm25_index is None:
            verses = self.indexer.load_bible_data()
            self.indexer.build_bm25_index(verses)
            
        # Ensure Qdrant has data
        collections = self.qdrant_client.get_collections()
        if self.collection_name not in [c.name for c in collections.collections]:
            verses = self.indexer.load_bible_data()
            self.indexer.index_verses(verses)
        
    def vector_search(
        self, 
        query: str, 
        top_k: int = config.DEFAULT_TOP_K,
        book_filter: Optional[str] = None
    ) -> List[Dict[str, Any]]:
        """
        Perform vector search using Qdrant.
        
        Args:
            query: Search query
            top_k: Number of results to return
            book_filter: Optional book name filter
            
        Returns:
            List of results with scores
        """
        # Generate query embedding
        query_embedding = self.embedding_model.encode(query)
        
        # Build filter if book is specified
        filter_condition = None
        if book_filter:
            filter_condition = Filter(
                must=[FieldCondition(
                    key="book",
                    match=MatchValue(value=book_filter)
                )]
            )
        
        # Search Qdrant
        results = self.qdrant_client.search(
            collection_name=self.collection_name,
            query_vector=query_embedding.tolist(),
            limit=top_k * 2,  # Get more for better fusion
            query_filter=filter_condition
        )
        
        # Format results
        formatted_results = []
        for result in results:
            formatted_results.append({
                'id': result.payload['id'],
                'book': result.payload['book'],
                'chapter': result.payload['chapter'],
                'verse': result.payload['verse'],
                'text': result.payload['text'],
                'testament': result.payload['testament'],
                'category': result.payload.get('category', 'Unknown'),
                'score': result.score
            })
            
        return formatted_results
    
    def bm25_search(
        self, 
        query: str, 
        top_k: int = config.DEFAULT_TOP_K,
        book_filter: Optional[str] = None
    ) -> List[Dict[str, Any]]:
        """
        Perform BM25 keyword search.
        
        Args:
            query: Search query
            top_k: Number of results to return
            book_filter: Optional book name filter
            
        Returns:
            List of results with scores
        """
        # Tokenize query
        try:
            nltk.data.find('tokenizers/punkt')
        except LookupError:
            nltk.download('punkt')
            
        tokenized_query = word_tokenize(query.lower())
        
        # Get BM25 scores
        scores = self.indexer._bm25_index.get_scores(tokenized_query)
        
        # Get all verses
        verses = self.indexer.load_bible_data()
        
        # Filter by book if specified
        if book_filter:
            filtered_indices = []
            for i, verse in enumerate(verses):
                if verse['book'] == book_filter:
                    filtered_indices.append(i)
        else:
            filtered_indices = list(range(len(verses)))
        
        # Get scores for filtered indices
        filtered_scores = []
        for idx in filtered_indices:
            filtered_scores.append({
                'index': idx,
                'score': scores[idx],
                'verse': verses[idx]
            })
        
        # Sort by score descending
        filtered_scores.sort(key=lambda x: x['score'], reverse=True)
        
        # Format results
        results = []
        for item in filtered_scores[:top_k]:
            verse = item['verse']
            results.append({
                'id': verse['id'],
                'book': verse['book'],
                'chapter': verse['chapter'],
                'verse': verse['verse'],
                'text': verse['text'],
                'testament': verse['testament'],
                'category': verse.get('category', 'Unknown'),
                'score': item['score']
            })
            
        return results
    
    def rrf_fusion(
        self,
        vector_results: List[Dict[str, Any]],
        bm25_results: List[Dict[str, Any]],
        top_k: int = config.DEFAULT_TOP_K
    ) -> List[Dict[str, Any]]:
        """
        Combine results using Reciprocal Rank Fusion.
        
        Args:
            vector_results: Results from vector search
            bm25_results: Results from BM25 search
            top_k: Number of final results to return
            
        Returns:
            Fused and sorted results
        """
        # Create rank maps
        rank_map = defaultdict(float)
        
        # Add vector search ranks
        for rank, result in enumerate(vector_results, start=1):
            result_id = result['id']
            rank_map[result_id] += 1 / (self.rrf_k + rank)
            
        # Add BM25 ranks
        for rank, result in enumerate(bm25_results, start=1):
            result_id = result['id']
            rank_map[result_id] += 1 / (self.rrf_k + rank)
            
        # Sort by score
        sorted_ids = sorted(rank_map.items(), key=lambda x: x[1], reverse=True)
        
        # Get full result data for top results
        # We need to merge data from vector results (which has full payload)
        id_to_result = {r['id']: r for r in vector_results}
        
        final_results = []
        for result_id, score in sorted_ids[:top_k]:
            if result_id in id_to_result:
                result = id_to_result[result_id].copy()
                result['rrf_score'] = score
                final_results.append(result)
            else:
                # Fallback: find in BM25 results
                for r in bm25_results:
                    if r['id'] == result_id:
                        result = r.copy()
                        result['rrf_score'] = score
                        final_results.append(result)
                        break
                        
        return final_results
    
    def search(
        self,
        query: str,
        top_k: int = config.DEFAULT_TOP_K,
        book_filter: Optional[str] = None,
        include_scores: bool = True
    ) -> List[Dict[str, Any]]:
        """
        Perform hybrid search combining vector and BM25 search.
        
        Args:
            query: Search query
            top_k: Number of results to return
            book_filter: Optional book name filter
            include_scores: Whether to include individual scores
            
        Returns:
            List of results with RRF scores
        """
        if not query.strip():
            return []
            
        logger.info(f"Performing hybrid search for: '{query}' (book: {book_filter})")
        
        # Perform both searches
        vector_results = self.vector_search(query, top_k=top_k*2, book_filter=book_filter)
        bm25_results = self.bm25_search(query, top_k=top_k*2, book_filter=book_filter)
        
        # Fuse results
        results = self.rrf_fusion(vector_results, bm25_results, top_k)
        
        # Optionally include individual scores
        if include_scores:
            for result in results:
                # Find corresponding vector score
                for v in vector_results:
                    if v['id'] == result['id']:
                        result['vector_score'] = v['score']
                        break
                # Find corresponding BM25 score
                for b in bm25_results:
                    if b['id'] == result['id']:
                        result['bm25_score'] = b['score']
                        break
                        
        logger.info(f"Found {len(results)} results")
        return results
    
    def get_books(self) -> List[str]:
        """
        Get list of all books in the dataset.
        
        Returns:
            Sorted list of book names
        """
        verses = self.indexer.load_bible_data()
        books = sorted(set(v['book'] for v in verses))
        return books

"""
Unified LLM provider for TBS with support for Ollama, Gemini, and OpenAI.

import logging
import json
from typing import Dict, Any, Optional, Generator, List
from abc import ABC, abstractmethod

import requests
import streamlit as st

# Setup logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class LLMProvider(ABC):
    """
    Abstract base class for LLM providers.
    """
    
    @abstractmethod
    def generate_stream(
        self,
        prompt: str,
        model_name: str,
        **kwargs
    ) -> Generator[str, None, None]:
        """
        Generate a streaming response from the LLM.
        
        Args:
            prompt: The prompt to send to the LLM
            model_name: Name of the model to use
            **kwargs: Additional provider-specific arguments
            
        Yields:
            Chunks of the response as they arrive
        """
        pass
        
    @abstractmethod
    def is_available(self) -> bool:
        """Check if the provider is available."""
        pass

class OllamaProvider(LLMProvider):
    """
    Ollama LLM provider for local model inference.
    """
    
    def __init__(self, host: str = config.OLLAMA_HOST):
        """
        Initialize the Ollama provider.
        
        Args:
            host: Ollama host URL
        """
        self.host = host
        
    def is_available(self) -> bool:
        """Check if Ollama is available."""
        try:
            response = requests.get(f"{self.host}/api/tags", timeout=5)
            return response.status_code == 200
        except:
            return False
            
    def generate_stream(
        self,
        prompt: str,
        model_name: str = config.DEFAULT_LLM_MODEL,
        temperature: float = 0.7,
        max_tokens: int = 1000,
        **kwargs
    ) -> Generator[str, None, None]:
        """
        Generate a streaming response from Ollama.
        
        Args:
            prompt: The prompt to send
            model_name: Model to use (defaults to configured default)
            temperature: Generation temperature
            max_tokens: Maximum tokens to generate
            **kwargs: Additional arguments
            
        Yields:
            Chunks of the response as they arrive
        """
        if not self.is_available():
            yield "⚠️ Ollama is not available. Please ensure Ollama is running and the model is installed."
            return
            
        try:
            url = f"{self.host}/api/generate"
            payload = {
                "model": model_name,
                "prompt": prompt,
                "stream": True,
                "options": {
                    "temperature": temperature,
                    "num_predict": max_tokens
                }
            }
            
            response = requests.post(url, json=payload, stream=True)
            response.raise_for_status()
            
            for line in response.iter_lines():
                if line:
                    try:
                        data = json.loads(line)
                        if 'response' in data:
                            yield data['response']
                        if data.get('done', False):
                            break
                    except json.JSONDecodeError:
                        continue
                        
        except Exception as e:
            logger.error(f"Ollama generation error: {e}")
            yield f"❌ Error generating response from Ollama: {str(e)}"

class GeminiProvider(LLMProvider):
    """
    Google Gemini LLM provider for cloud inference.
    """
    
    def __init__(self, api_key: Optional[str] = None):
        """
        Initialize the Gemini provider.
        
        Args:
            api_key: Gemini API key (defaults to config)
        """
        self.api_key = api_key or config.GEMINI_API_KEY
        
    def is_available(self) -> bool:
        """Check if Gemini API is available."""
        return bool(self.api_key)
        
    def generate_stream(
        self,
        prompt: str,
        model_name: str = "gemini-2.0-flash-exp",
        temperature: float = 0.7,
        max_tokens: int = 1000,
        **kwargs
    ) -> Generator[str, None, None]:
        """
        Generate a streaming response from Gemini.
        
        Args:
            prompt: The prompt to send
            model_name: Model to use (defaults to gemini-2.0-flash-exp)
            temperature: Generation temperature
            max_tokens: Maximum tokens to generate
            **kwargs: Additional arguments
            
        Yields:
            Chunks of the response as they arrive
        """
        if not self.is_available():
            yield "⚠️ Gemini API key is not configured. Please set GEMINI_API_KEY in your environment."
            return
            
        try:
            import google.generativeai as genai
            
            genai.configure(api_key=self.api_key)
            
            model = genai.GenerativeModel(
                model_name=model_name,
                generation_config={
                    'temperature': temperature,
                    'max_output_tokens': max_tokens,
                    'top_p': 0.95,
                    'top_k': 40
                }
            )
            
            response = model.generate_content(prompt, stream=True)
            
            for chunk in response:
                if chunk.text:
                    yield chunk.text
                    
        except ImportError:
            yield "❌ google-generativeai package not installed. Run: uv add google-generativeai"
        except Exception as e:
            logger.error(f"Gemini generation error: {e}")
            yield f"❌ Error generating response from Gemini: {str(e)}"

class OpenAIProvider(LLMProvider):
    """
    OpenAI LLM provider for cloud inference.
    """
    
    def __init__(self, api_key: Optional[str] = None):
        """
        Initialize the OpenAI provider.
        
        Args:
            api_key: OpenAI API key (defaults to config)
        """
        self.api_key = api_key or config.OPENAI_API_KEY
        
    def is_available(self) -> bool:
        """Check if OpenAI API is available."""
        return bool(self.api_key)
        
    def generate_stream(
        self,
        prompt: str,
        model_name: str = "gpt-4o-mini",
        temperature: float = 0.7,
        max_tokens: int = 1000,
        **kwargs
    ) -> Generator[str, None, None]:
        """
        Generate a streaming response from OpenAI.
        
        Args:
            prompt: The prompt to send
            model_name: Model to use (defaults to gpt-4o-mini)
            temperature: Generation temperature
            max_tokens: Maximum tokens to generate
            **kwargs: Additional arguments
            
        Yields:
            Chunks of the response as they arrive
        """
        if not self.is_available():
            yield "⚠️ OpenAI API key is not configured. Please set OPENAI_API_KEY in your environment."
            return
            
        try:
            from openai import OpenAI
            
            client = OpenAI(api_key=self.api_key)
            
            response = client.chat.completions.create(
                model=model_name,
                messages=[
                    {"role": "system", "content": self.get_system_prompt()},
                    {"role": "user", "content": prompt}
                ],
                temperature=temperature,
                max_tokens=max_tokens,
                stream=True
            )
            
            for chunk in response:
                if chunk.choices[0].delta.content is not None:
                    yield chunk.choices[0].delta.content
                    
        except ImportError:
            yield "❌ openai package not installed. Run: uv add openai"
        except Exception as e:
            logger.error(f"OpenAI generation error: {e}")
            yield f"❌ Error generating response from OpenAI: {str(e)}"
            
    def get_system_prompt(self) -> str:
        """Get the system prompt for Bible exegesis."""
        return """You are TBS (Tuixiu Bible Search), a precise, reverent, and scholarly biblical assistant. 
Answer user queries strictly grounded in the retrieved passages provided. 
Always cite Book, Chapter, and Verse numbers.
Be respectful of the sacred text and provide thoughtful, well-reasoned responses.
When appropriate, provide historical and cultural context for the passages discussed."""

class LLMProviderFactory:
    """
    Factory for creating LLM providers.
    """
    
    @staticmethod
    def create_provider(provider_type: str) -> LLMProvider:
        """
        Create an LLM provider instance.
        
        Args:
            provider_type: Type of provider ('ollama', 'gemini', 'openai')
            
        Returns:
            LLMProvider instance
        """
        providers = {
            'ollama': OllamaProvider,
            'gemini': GeminiProvider,
            'openai': OpenAIProvider
        }
        
        provider_class = providers.get(provider_type)
        if not provider_class:
            raise ValueError(f"Unknown provider type: {provider_type}")
            
        return provider_class()
        
    @staticmethod
    def get_available_providers() -> List[str]:
        """
        Get list of available providers.
        
        Returns:
            List of provider types that are available
        """
        available = []
        
        # Check Ollama
        try:
            ollama = OllamaProvider()
            if ollama.is_available():
                available.append('ollama')
        except:
            pass
            
        # Check Gemini
        gemini = GeminiProvider()
        if gemini.is_available():
            available.append('gemini')
            
        # Check OpenAI
        openai = OpenAIProvider()
        if openai.is_available():
            available.append('openai')
            
        return available
        
    @staticmethod
    def get_provider_models(provider_type: str) -> List[str]:
        """
        Get list of models available for a provider.
        
        Args:
            provider_type: Type of provider
            
        Returns:
            List of model names
        """
        models = {
            'ollama': ['llama3', 'phi4', 'mistral', 'gemma2', 'llama3.1', 'mixtral'],
            'gemini': ['gemini-2.0-flash-exp', 'gemini-1.5-pro', 'gemini-1.5-flash'],
            'openai': ['gpt-4o-mini', 'gpt-4o', 'gpt-4-turbo', 'gpt-3.5-turbo']
        }
        
        return models.get(provider_type, [])

# 📖 TBS - Tuixiu Bible Search

A hybrid, local-first Bible search and study assistant combining semantic vector search, BM25 keyword search, and AI-powered synthesis.

## ✨ Features

- **Hybrid Search**: Combines semantic understanding with precise keyword matching using Reciprocal Rank Fusion (RRF)
- **Local-first**: Runs locally with optional cloud LLM integration
- **Multiple LLM Providers**: Support for Ollama (local), Gemini, and OpenAI
- **Modern UI**: Clean, responsive Streamlit interface with verse cards and metadata
- **AI Synthesis**: Generate scholarly exegesis using retrieved passages
- **Book Filtering**: Search within specific books of the Bible
- **Export Results**: Download search results as CSV

🔍 How It Works
Indexing:

Verses are loaded from data/sample_bible.json

Embeddings are generated using all-MiniLM-L6-v2

BM25 index is built using rank-bm25

Vectors are stored in Qdrant (in-memory)

Search:

Query is processed by both vector (Qdrant) and BM25 search

Results from both methods are fused using RRF

Final ranked results are displayed with metadata

AI Synthesis:

Retrieved passages are used as context

Selected LLM provider generates a scholarly synthesis

Streaming responses for real-time feedback

Search Settings
Number of Results: 3-20 verses per search

RRF k: Ranking constant (default: 60)

LLM Settings
Provider: Ollama, Gemini, or OpenAI

Temperature: 0.0-1.0 for response creativity
